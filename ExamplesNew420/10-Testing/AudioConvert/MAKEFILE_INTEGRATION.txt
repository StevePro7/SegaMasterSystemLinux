# 5. COMPLETE MAKEFILE EXAMPLE

# Here's how it might look in your Makefile:

# --- Variables ---
BUILD = build
TARGET = $(BUILD)/transbot.sms

SOURCES = \
	dev/main.c \
	dev/engine/audio_manager.c

OBJECTS = $(SOURCES:.c=.o)
OBJECTS := $(OBJECTS:%=$(BUILD)/%)

CFLAGS = -Idev/engine/ -c -O2
ASFLAGS = -i dev/engine/

AUDIO_DATA_H = dev/engine/audio_data.h

# --- Targets ---
all: $(AUDIO_DATA_H) $(TARGET)

$(TARGET): $(OBJECTS)
	sdcc -o $@ $(OBJECTS)

$(BUILD)/%.o: %.c $(AUDIO_DATA_H)
	mkdir -p $(dir $@)
	sdcc $(CFLAGS) -o $@ $<

$(AUDIO_DATA_H): TransbotASM/content/psg_data.inc extract_audio_data.py
	python3 extract_audio_data.py TransbotASM/content/psg_data.inc $@

clean:
	rm -rf $(BUILD)/
	rm -f $(AUDIO_DATA_H)

.PHONY: all clean
